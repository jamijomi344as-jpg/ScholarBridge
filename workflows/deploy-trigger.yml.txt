name: Trigger n8n Agent on Deploy

on:
  push:
    branches:
      - main  # Agar asosiy branch'ingiz 'master' bo'lsa, buni 'master' qiling

jobs:
  notify-n8n:
    runs-on: ubuntu-latest
    steps:
      - name: Wait for Render deploy to finish
        run: sleep 60  # Render saytni build qilib tayyor bo'lishi uchun 60 sek kutadi

      - name: Call n8n Webhook
        run: |
          curl -X POST https://hushnudsagent.app.n8n.cloud/webhook/render-deploy \
               -H "Content-Type: application/json" \
               -d '{"event": "deploy_success", "source": "github_actions"}'
